% Set system limits
lims = mr.opts('MaxGrad', 20, 'gradUnit', 'mT/m',...
              'MaxSlew', 35, 'SlewUnit', 'T/m/s', ...
              'rfRingdownTime', 20e-6, ...
              'rfDeadTime', 100e-6, ...
              'adcDeadTime', 10e-6, ...
              'B0', 0.55);
seq=mr.Sequence();              % Create a new sequence object
fov=256e-3; Nx=128; Ny=128;       % Define FOV and resolution
slice_thickness = 10e-3;          % Grosor de la slice
TE=10e-3;
flip_angle = 90*pi/180;
[rf, gz] = mr.makeSincPulse(flip_angle,lims,'Duration',3e-3,...
    'SliceThickness',slice_thickness,'apodization',0.5,'timeBwProduct',4);

% Define other gradients and ADC events
deltak=1/fov; %viene del NUT=1
kWidth = Nx*deltak;
readoutTime = 12.8e-4;
gx = mr.makeTrapezoid('x',lims,'FlatArea',kWidth,'FlatTime',readoutTime);
adc = mr.makeAdc(Nx,lims,'Duration',gx.flatTime,'Delay',gx.riseTime);

% Pre-phasing gradients
preTime=2e-3;
gzReph = mr.makeTrapezoid('z',lims,'Area',-gz.area/2,'Duration',preTime);
gxPre = mr.makeTrapezoid('x',lims,'Area',-gx.area/2-deltak/2,'Duration',preTime);
gyPre = mr.makeTrapezoid('y',lims,'Area',-Ny/2*deltak,'Duration',preTime);

delayTE = TE;
% Define sequence blocks

amplitud=gx.amplitude;
for shot=1:Ny
    gyPre = mr.makeTrapezoid('y',lims,'Area',-Ny/2*deltak+(shot-1)*deltak,'Duration',preTime);
    seq.addBlock(rf,gz);
    seq.addBlock(gxPre,gyPre,gzReph);
    seq.addBlock(mr.makeDelay(delayTE));
    seq.addBlock(gx,adc);           % Read one line of k-space
    seq.addBlock(mr.makeDelay(0.25)); %TR
end

seq.setDefinition('FOV', [fov fov slice_thickness]);
seq.write('gre_1.seq');   % Output sequence for scanner
seq.plot();             % Plot sequence waveforms

rep = seq.testReport;
fprintf([rep{:}]);
% %% Checking k-space
[ktraj_adc, t_adc, ktraj, t_ktraj, t_excitation, t_refocusing, ~, ~] = seq.calculateKspacePP();
figure;
plot3(ktraj_adc(1,:), ktraj_adc(2,:), t_adc*1e3)
xlabel('kx')
ylabel('ky')
zlabel('time')
title('Timemap')
%%

plot(ktraj_adc(1,:), ktraj_adc(2,:))
